// ============ IndexedDB 存储（PDF + 练习数据） ============ const DB_NAME = 'competition_files_db'; const DB_VERSION = 1; let dbReady = null; let dbConnection = null; // 单例长连接，不会重复创建 function openDB() { if (dbReady) return dbReady; dbReady = new Promise((resolve, reject) => { const req = indexedDB.open(DB_NAME, DB_VERSION); req.onupgradeneeded = (e) => { const db = e.target.result; if (!db.objectStoreNames.contains('files')) { db.createObjectStore('files', { keyPath: 'name' }); } }; req.onsuccess = (e) => { dbConnection = e.target.result; // 连接意外关闭时清除缓存 dbConnection.onclose = () => { dbReady = null; dbConnection = null; }; dbConnection.onversionchange = () => { dbConnection.close(); dbReady = null; dbConnection = null; }; resolve(dbConnection); }; req.onerror = (e) => reject(e.target.error); req.onblocked = () => { console.warn('⏳ IndexedDB 被其他标签页阻塞，等待中...'); }; // 超时保护 setTimeout(() => { if (req.readyState === 'pending') { reject(new Error('IndexedDB 打开超时')); } }, 5000); }); return dbReady; } async function dbPutFile(name, blob, type) { const db = await openDB(); console.log('💾 dbPutFile:', name, '| size:', (blob.size / 1024 / 1024).toFixed(1) + 'MB', '| type:', type); return new Promise((resolve, reject) => { const tx = db.transaction('files', 'readwrite'); const store = tx.objectStore('files'); store.put({ name, blob, type, addedAt: Date.now() }); tx.oncomplete = () => { console.log(' ✅ 写入完成:', name); resolve(); }; tx.onerror = () => { console.error(' ❌ 写入失败:', name, tx.error); reject(tx.error); }; }); } async function dbGetFile(name) { const db = await openDB(); return new Promise((resolve, reject) => { const tx = db.transaction('files', 'readonly'); const store = tx.objectStore('files'); const req = store.get(name); req.onsuccess = () => resolve(req.result || null); req.onerror = () => reject(req.error); }); } async function dbHasFile(name) { const f = await dbGetFile(name); return !!f; } async function dbGetAllFileNames() { const db = await openDB(); return new Promise((resolve, reject) => { const tx = db.transaction('files', 'readonly'); const store = tx.objectStore('files'); const req = store.getAllKeys(); req.onsuccess = () => resolve(req.result || []); req.onerror = () => reject(req.error); }); } async function dbDeleteFile(name) { const db = await openDB(); console.log('🗑️ dbDeleteFile:', name); return new Promise((resolve, reject) => { const tx = db.transaction('files', 'readwrite'); const store = tx.objectStore('files'); store.delete(name); tx.oncomplete = () => { console.log(' ✅ 已删除:', name); resolve(); }; tx.onerror = () => { console.error(' ❌ 删除失败:', name, tx.error); reject(tx.error); }; }); }
